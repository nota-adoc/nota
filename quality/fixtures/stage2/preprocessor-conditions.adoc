= Conditional directives
:a: 1
:b: 2

== Block forms

ifdef::a[]
* a is set
endif::[]

ifdef::missing[]
* missing is set
endif::[]

ifndef::a[]
* a is not set
endif::[]

ifndef::missing[]
* missing is not set
endif::[]

ifdef::a,missing[]
* a or missing
endif::[]

ifdef::a+missing[]
* a and missing
endif::[]

ifdef::a+b[]
* a and b
endif::[]

ifndef::a,missing[]
* neither a nor missing (a is set)
endif::[]

ifndef::a+missing[]
* not both a and missing
endif::[]

ifdef::A[]
* names are case insensitive
endif::[]

ifdef::a[]

== Eval

ifeval::[{a} == 1]
* a equals 1
endif::[]

ifeval::[{a} > 1]
* a above 1
endif::[]

ifeval::[{b} >= 2]
* b at least 2
endif::[]

ifeval::[{a} != {b}]
* a differs from b
endif::[]

ifeval::["{a}" == "1"]
* string compare
endif::[]

ifeval::['{a}' == '1']
* single quotes
endif::[]

ifeval::[{a} < 0.5]
* float compare
endif::[]

ifeval::[true == true]
* booleans
endif::[]

ifeval::["x" < "y"]
* strings order
endif::[]
